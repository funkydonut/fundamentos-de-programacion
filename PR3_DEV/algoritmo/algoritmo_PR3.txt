const
    MAX_NODES: integer = 20 {Maximum number of nodes}
    SECTORS_PER_NODE: integer = 5; {Number of sectors per node}
    OVERLOAD_TOTAL_THRESHOLD: real = 500.0 {Limit for overload consumption}
    CRITICAL_SECTOR_LIMIT: real = 150.0 {Limit for security sector consumption}
    KW_TO_MW: real = 1000.0; {kW to MW divisor factor}
end const

type
    {Enum to represent sector types}
    tSectorType = {INDUSTRY, RESIDENTIAL, TRANSPORT, SERVICES, SANITARY}

    {Available power scales}
    tPowerScale = {KW, MW};

    tSectorData = record {Structure to represent a sector's data}
        sector: tSectorType; {Type of sector}
        currentConsumption: real; {Consumption in kW }
    end record

    tGridNode = record {Structure to represent a node info}
        nodeName: string; {Name of the node}
        {Vector of sectors within the node}
        sectorList: vector[SECTORS_PER_NODE] of tSectorData;
    end record

    tGridTable = record {Structure to represent a grid}
        {Vector of nodes within the grid}
        nodes: vector[MAX_NODES] of tGridNode;
        nNodes: integer; {Number of nodes in the grid}
    end record
end type

{Exercise 3.1}
{Action to print a formatted report of all nodes}
action writeGridReport(in grid: tGridTable, in powerScale: tPowerScale)
    var
        i: integer;
        minNodeConsumption: real;
        maxNodeConsumption: real;
    end var

    writeString("GRID NODES:");
    writeString("NODE_NAME         MIN_CONS      MAX_CONS");
    writeString("=============================================");
    for i := 1 to grid.nNodes do
        getExtremConsForecast(grid.nodes[i], minNodeConsumption, maxNodeConsumption);
        if powerScale = MW then
            minNodeConsumption := kwToMw(minNodeConsumption);
            maxNodeConsumption := kwToMw(maxNodeConsumption);
        end if
        writeString(grid.nodes[i].nodeName);
        writeReal(minNodeConsumption);
        writeReal(maxNodeConsumption);
    end for
end action

{Exercise 4}
{Action to find min and max consumption in a node}
action getExtremConsForecast(in node: tGridNode, out minNodeConsumption: real, out maxNodeConsumption: real)
    var
        i: integer;
    end var

    minNodeConsumption := node.sectorList[1].currentConsumption;
    maxNodeConsumption := node.sectorList[1].currentConsumption;
    for i := 2 to SECTORS_PER_NODE do
        if node.sectorList[i].currentConsumption < minNodeConsumption then
            minNodeConsumption := node.sectorList[i].currentConsumption;
        else
            if node.sectorList[i].currentConsumption > maxNodeConsumption then
                maxNodeConsumption := node.sectorList[i].currentConsumption;
            end if
        end if
    end for
end action

{Exercise 5}
{Function to convert kilowatts to megawatts}
function kwToMw(kw: real): real
    return kw / KW_TO_MW;
end function

{Exercise 6}
{Action to filter overloaded nodes into a new table}
action getOverloadedNodes(in grid: tGridTable, out overloadedGrid: tGridTable)
    var
        i: integer;
    end var

    overloadedGrid.nNodes := 0;
    for i := 1 to grid.nNodes do
        if isOverloaded(grid.nodes[i]) then
            overloadedGrid.nodes[overloadedGrid.nNodes + 1] := grid.nodes[i];
            overloadedGrid.nNodes := overloadedGrid.nNodes + 1;
        end if
    end for
end action

{Exercise 8}
{Action that removes nodes with 0.0kW max consumption}
action removeInactiveNodes(inout grid: tGridTable)
    var
        minConsumption: real;
        maxConsumption: real;
        i: integer;
        j: integer;
    end var

    for i := 1 to grid.nNodes do
        getExtremConsForecast(grid.nodes[i], minConsumption, maxConsumption);
        if maxConsumption ≤ 0.0 then
            grid.nNodes := grid.nNodes - 1;
            for j := i to grid.nNodes do
                grid.nodes[j] := grid.nodes[j + 1];
            end for
            i := i - 1;
        end if
    end for
end action

{Action for load data from file}
action loadGridDataFromFile(in filename: string, out grid: tGridTable, out isRead: boolean)
    var
        fileToRead: file;
        newNode: tGridNode;
        i: integer;
    end var

    {Open the input file}
    fileToRead := openFile(filename);
    if fileToRead ≠ NULL then
        grid.nNodes := 0;
        {Read while there are nodes and space available}
        while not isEndOfFile(fileToRead) and grid.nNodes < MAX_NODES do
            {Read node name}
            newNode.nodeName := readStringFromFile(fileToRead);
            {Read all sector data of this node}
            for i := 1 to SECTORS_PER_NODE do
                {Read sector type and consumption}
                newNode.sectorList[i].sector := readEnumFromFile(fileToRead);
                newNode.sectorList[i].currentConsumption := readRealFromFile(fileToRead);
            end for
            {Add new node to grid table}
            grid.nNodes := grid.nNodes + 1;
            grid.nodes[grid.nNodes] := newNode;
        end while
        {Close the file}
        closeFile(fileToRead);
        isRead := true;
    else
        isRead := false;
    end if
end action

{Function to check for overloads}
function isOverloaded(node: tGridNode): boolean
    var
        total: real;
        criticalOverload: boolean;
        i: integer;
    end var

    total := 0.0;
    criticalOverload := false;
    i := 1;
    while i ≤ SECTORS_PER_NODE and not criticalOverload do
        total := total + node.sectorList[i].currentConsumption;
        {Node is in critical overload if any sector consumes more than 150kW
        or the total accumulated consumption exceeds 500kW}
        criticalOverload := total > OVERLOAD_TOTAL_THRESHOLD or
                            node.sectorList[i].currentConsumption > CRITICAL_SECTOR_LIMIT;
        i := i + 1;
    end while
    return criticalOverload;
end function

algorithm UOCSmartGridManagement
    var
        grid: tGridTable; {Node table}
        isRead: boolean; {Result of reading}
        i: integer;
        filename: string; {Name of file to read}
        powerScale: tPowerScale; {Power scale selected by the user}
        overloadedGrid: tGridTable; {Overloaded nodes table}
        originalNodes: integer;
        {...}
    end var

    {Exercise 1}
    {Data input and file loading}
    writeString("INPUT DATA");
    writeString("LOAD DATA FROM FILE. ENTER FILE NAME >> ");
    filename := readString();
    loadGridDataFromFile(filename, grid, isRead);

    {Only proceed if the file was read successfully and is not empty}
    if isRead and grid.nNodes > 0 then
        {Exercise 2}
        {Power scale input with validation}
        writeString("SELECT POWER SCALE (1-KW, 2-MW) >> ");
        powerScale := readInteger();
        while powerScale ≠ KW and powerScale ≠ MW do
            writeString("INVALID DATA, TRY AGAIN!");
            powerScale := readInteger();
        end while

        {Exercise 3}
        {General network report}
        writeGridReport(grid, powerScale);

        {Exercise 6}
        {Critical nodes filtering and alerts}
        writeString("OVERLOADED NODES REPORT");
        writeString("=======================");
        getOverloadedNodes(grid, overloadedGrid);

        {Exercise 7}
        {Show nodes with critical alert, if exists any}
        if overloadedGrid.nNodes > 0 then
            for i := 1 to overloadedGrid.nNodes do
                writeString("CRITICAL ALERT: ");
                writeString(overloadedGrid.nodes[i].nodeName);
            end for
        else
            writeString("ALL NODES OPERATING NORMALLY.");
        end if

        {Exercise 8}
        {Network maintenance: cleanup of inactive nodes}
        originalNodes := grid.nNodes;
        removeInactiveNodes(grid);

        {Exercise 8.1}
        {Check if any node was removed and show clean table}
        if originalNodes > grid.nNodes then
            writeString("INACTIVE NODES REMOVED: ");
            writeInteger(originalNodes - grid.nNodes);
            writeGridReport(grid, powerScale);
        else
            writeString("NO INACTIVE NODES");
        end if
    else
        {Error message if loading failed}
        writeString("ERROR: COULD NOT READ FILE OR TABLE EMPTY");
    end if
end algorithm
